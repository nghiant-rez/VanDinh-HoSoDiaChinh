CREATE TABLE [dbo].[LichSuHoSo] ( 
    [Id] BIGINT IDENTITY(1,1) PRIMARY KEY, 
    [HoSoId] BIGINT NOT NULL, 
    [Action] NVARCHAR(100) NOT NULL,

    [PerformedByUserId] BIGINT NOT NULL, 
    [PerformedAt] DATETIME2 DEFAULT GETDATE(),

    FOREIGN KEY ([PerformedByUserId]) REFERENCES [dbo].Users

-- ============================================================================== 
-- PHẦN 7: TRIGGER AUDIT LOG CHO HỒ SƠ 
-- ============================================================================== 
CREATE TRIGGER [dbo].[TRG_HoSo_AuditLog] 
ON [dbo].[HoSo] 
AFTER INSERT, UPDATE 
AS 
BEGIN 
    SET NOCOUNT ON;
    DECLARE @Action NVARCHAR(100);

    IF EXISTS (SELECT * FROM inserted) AND EXISTS (SELECT * FROM deleted)
        SET @Action = 'UPDATE';
    ELSE IF EXISTS (SELECT * FROM inserted)
        SET @Action = 'INSERT';
    ELSE
        RETURN; 

    DECLARE @PerformedByUserId BIGINT;
    SET @PerformedByUserId = CAST(SESSION_CONTEXT(N'UserId') AS BIGINT);

    IF @PerformedByUserId IS NULL
    BEGIN
        SELECT TOP 1 @PerformedByUserId = CreatedByUserId FROM inserted;
    END

    IF @Action = 'UPDATE'
    BEGIN
        INSERT INTO [dbo].[LichSuHoSo] (
            [HoSoId], [Action], [PerformedByUserId], [PerformedAt], [OldValue], [NewValue]
        )
        SELECT 
            i.Id, 
            'UPDATE', 
            @PerformedByUserId, 
            GETDATE(),
            (SELECT d.Id, d.IdCha, d.MaHoSo, d.TenHoSo, d.LoaiHoSoId, d.ThuaDatId, d.DuAnId, d.KhoLuuTruId, d.KeLuuTruId, d.TangLuuTruId, d.HopSoLuuTruId, d.AdditionalInfo, d.MapCoordinates.ToString() AS MapCoordinatesWKT FROM deleted d WHERE d.Id = i.Id FOR JSON PATH, WITHOUT_ARRAY_WRAPPER),
            (SELECT i2.Id, i2.IdCha, i2.MaHoSo, i2.TenHoSo, i2.LoaiHoSoId, i2.ThuaDatId, i2.DuAnId, i2.KhoLuuTruId, i2.KeLuuTruId, i2.TangLuuTruId, i2.HopSoLuuTruId, i2.AdditionalInfo, i2.MapCoordinates.ToString() AS MapCoordinatesWKT FROM inserted i2 WHERE i2.Id = i.Id FOR JSON PATH, WITHOUT_ARRAY_WRAPPER)
        FROM inserted i
        INNER JOIN deleted d ON i.Id = d.Id;
    END
    ELSE IF @Action = 'INSERT'
    BEGIN
         INSERT INTO [dbo].[LichSuHoSo] (
            [HoSoId], [Action], [PerformedByUserId], [PerformedAt], [OldValue], [NewValue]
        )
        SELECT 
            i.Id, 
            'INSERT', 
            @PerformedByUserId, 
            GETDATE(),
            NULL, 
            (SELECT i2.Id, i2.IdCha, i2.MaHoSo, i2.TenHoSo, i2.LoaiHoSoId, i2.ThuaDatId, i2.DuAnId, i2.KhoLuuTruId, i2.KeLuuTruId, i2.TangLuuTruId, i2.HopSoLuuTruId, i2.AdditionalInfo, i2.MapCoordinates.ToString() AS MapCoordinatesWKT FROM inserted i2 WHERE i2.Id = i.Id FOR JSON PATH, WITHOUT_ARRAY_WRAPPER)
        FROM inserted i;
    END
END
